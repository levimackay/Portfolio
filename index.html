    fflush(stdout);

static long rt_add(long a, long b, const char *at) {
    long r;
    if (__builtin_add_overflow(a, b, &r)) rt_fail("integer overflow", at);
    return r;
}

    return r;
}

    long r;

static long rt_div(long a, long b, const char *at) {
    if (b == 0) rt_fail("division by zero", at);
    if (a == LONG_MIN && b == -1) rt_fail("integer overflow", at);
    return a / b;
}

static long rt_mod(long a, long b, const char *at) {
    if (b == 0) rt_fail("division by zero", at);
    if (a == LONG_MIN && b == -1) rt_fail("integer overflow", at);
    return a % b;
}

static long rt_neg(long a, const char *at) {
    if (a == LONG_MIN) rt_fail("integer overflow", at);
    return -a;
}

static long rt_leave(long value) {
    rt_depth--;
    return value;
}

static void rt_print_int(long value) {
    printf("%ld\n", value);
}

static void rt_print_bool(bool value) {
    puts(value ? "true" : "false");
}

static void rt_enter(const char *at) {
    if (++rt_depth > 10000) rt_fail("more than 10000 nested calls", at);
}

static long iz_fib(long iz_n) {
    rt_enter("playground.iz:1:4");
    if (iz_n < 2) {
        return rt_leave(iz_n);
    }
    long t1 = iz_fib(rt_sub(iz_n, 1, "playground.iz:5:18"));
    return rt_leave(rt_add(t1, iz_fib(rt_sub(iz_n, 2, "playground.iz:5:31")), "playground.iz:5:23"));
}

int main(void) {

        iz_i = rt_add(iz_i, 1, "playground.iz:11:11");
